# COMANDOS PARA EJECUTAR EN LA TERMINAL DE INTELLIJ IDEA
# Copiar y pegar cada bloque en orden

# ========================================
# 1. CONFIG SERVICE (Puerto 8888)
# ========================================
cd config-service
.\mvnw.cmd spring-boot:run

# Esperar a que aparezca "Started ConfigServiceApplication" en la consola
# Luego abrir nueva terminal (Ctrl+Shift+T) y continuar con el siguiente servicio

# ========================================
# 2. EUREKA SERVICE (Puerto 8761)
# ========================================
cd eureka-service
.\mvnw.cmd spring-boot:run

# Esperar a que aparezca "Started EurekaServiceApplication" en la consola
# Luego abrir nueva terminal y continuar

# ========================================
# 3. GATEWAY SERVICE (Puerto 8080)
# ========================================
cd gateway-service
.\mvnw.cmd spring-boot:run

# Esperar a que aparezca "Started GatewayServiceApplication" en la consola
# Luego abrir nueva terminal y continuar

# ========================================
# 4. IAM SERVICE (Puerto 8081)
# ========================================
cd iam-service
.\mvnw.cmd spring-boot:run

# Esperar a que aparezca "Started IamServiceApplication" en la consola
# Luego abrir nueva terminal y continuar

# ========================================
# 5. POST SERVICE (Puerto 8082)
# ========================================
cd post-service
.\mvnw.cmd spring-boot:run

# Esperar a que aparezca "Started PostServiceApplication" en la consola
